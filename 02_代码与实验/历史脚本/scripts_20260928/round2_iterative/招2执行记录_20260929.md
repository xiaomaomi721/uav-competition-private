cd /data/uav

    --output-dir teacher_cache/treat_scores 2>&1 | tee teacher_cache/treat_gen.log

CUDA_VISIBLE_DEVICES=1 venv/bin/python code/package_treat_round2.py \
    --checkpoint runs/round2_treat/checkpoint_step2000.pt \
